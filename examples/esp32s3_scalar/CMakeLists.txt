cmake_minimum_required(VERSION 3.22)

if(NOT DEFINED GE_PROJECT_ROOT)
    get_filename_component(GE_PROJECT_ROOT "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
endif()
if(NOT EXISTS "${GE_PROJECT_ROOT}/examples/leaf.goose")
    message(FATAL_ERROR "GE_PROJECT_ROOT must identify the Goose Embedded repository")
endif()
if(NOT DEFINED GOOSE_COMPILER OR NOT EXISTS "${GOOSE_COMPILER}")
    message(FATAL_ERROR "Set GOOSE_COMPILER to the prepared scalar-v0 host compiler")
endif()
set(GE_SCALAR_OPT "0" CACHE STRING "Goose and component optimization level (0 or 2)")
if(NOT GE_SCALAR_OPT MATCHES "^(0|2)$")
    message(FATAL_ERROR "GE_SCALAR_OPT must be 0 or 2")
endif()
set(PROJECT_VER "0.1.0")
set(GE_SCALAR_CASE "success" CACHE STRING "Firmware case: success, cold-export, overflow")
if(NOT GE_SCALAR_CASE MATCHES "^(success|cold-export|overflow)$")
    message(FATAL_ERROR "GE_SCALAR_CASE must be success, cold-export or overflow")
endif()
include($ENV{IDF_PATH}/tools/cmake/project.cmake)
idf_build_set_property(MINIMAL_BUILD ON)
project(goose_scalar)
if(NOT IDF_TARGET STREQUAL "esp32s3")
    message(FATAL_ERROR "This proof of concept supports only esp32s3")
endif()
